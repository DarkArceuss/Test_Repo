-- ==================== FILE IO ====================

local hasFileIO = (type(writefile) == "function") and (type(readfile) == "function") and (type(isfile) == "function")

local function saveFile(name, data)
	if not hasFileIO then return end
	pcall(function() writefile(name, data) end)
end
local function readFile(name)
	if not hasFileIO then return nil end
	if not isfile(name) then return nil end
	local ok, data = pcall(readfile, name)
	if ok then return data end
	return nil
end

-- ==================== MEMORY ====================

local memory = {
	summary = "",
	facts = {},
	recent = {},
	savedAt = os.time and os.time() or 0,
}

local function loadMemory()
	local raw = readFile(MEMORY_FILE)
	if raw and raw ~= "" then
		local ok, data = pcall(function() return HttpService:JSONDecode(raw) end)
		if ok and type(data) == "table" then
			memory.summary = data.summary or ""
			memory.facts = data.facts or {}
			memory.recent = data.recent or {}
			memory.savedAt = data.savedAt or 0
		end
	end
end

local function saveMemory()
	memory.savedAt = os.time and os.time() or 0
	local ok, enc = pcall(function() return HttpService:JSONEncode(memory) end)
	if ok then saveFile(MEMORY_FILE, enc) end
end

-- ==================== HELPERS ====================

local function getChar()
	local char = LocalPlayer and LocalPlayer.Character
	if not char then return nil end
	return char, char:FindFirstChildOfClass("Humanoid"), char:FindFirstChild("HumanoidRootPart")
end
local function getHum() local _, h = getChar() return h end
local function getHRP() local _, _, r = getChar() return r end

local function resolvePath(path)
	if not path or type(path) ~= "string" then return nil end
	path = path:gsub("^game%.", ""):gsub("^game:", "")
	local parts = {}
	for part in path:gmatch("[^%.]+") do table.insert(parts, part) end
	local current = game
	for _, part in ipairs(parts) do
		if current == game then
			local svc = game:FindFirstChild(part)
			if svc then current = svc
			else
				local ok, s = pcall(function() return game:GetService(part) end)
				if ok and s then current = s else return nil end
			end
		else
			local n = current:FindFirstChild(part)
			if not n then return nil end
			current = n
		end
	end
	return current
end

local function getPath(obj)
	if not obj then return "nil" end
	local ok, full = pcall(function() return obj:GetFullName() end)
	return ok and full or obj.Name
end

local CHARACTER_ACTIONS = {
	set_speed = function(p)
		local h = getHum(); if not h then return false, "no character" end
		h.WalkSpeed = tonumber(p.value) or h.WalkSpeed
		return true, "WalkSpeed = " .. tostring(h.WalkSpeed)
	end,
	set_jump = function(p)
		local h = getHum(); if not h then return false, "no character" end
		h.UseJumpPower = true; h.JumpPower = tonumber(p.value) or h.JumpPower
		return true, "JumpPower = " .. tostring(h.JumpPower)
	end,
	set_hip_height = function(p)
		local h = getHum(); if not h then return false, "no character" end
		h.HipHeight = tonumber(p.value) or h.HipHeight
		return true, "HipHeight = " .. tostring(h.HipHeight)
	end,
	teleport = function(p)
		local r = getHRP(); if not r then return false, "no character" end
		local x = tonumber(p.x) or r.Position.X
		local y = tonumber(p.y) or r.Position.Y
		local z = tonumber(p.z) or r.Position.Z
		r.CFrame = CFrame.new(x, y, z)
		return true, string.format("Teleport to (%.1f, %.1f, %.1f)", x, y, z)
	end,
	get_position = function()
		local r = getHRP(); if not r then return false, "no character" end
		local p = r.Position
		return true, string.format("Position: (%.1f, %.1f, %.1f)", p.X, p.Y, p.Z)
	end,
	heal = function()
		local h = getHum(); if not h then return false, "no character" end
		h.Health = h.MaxHealth
		return true, "Health restored"
	end,
	set_health = function(p)
		local h = getHum(); if not h then return false, "no character" end
		local v = tonumber(p.value); if not v then return false, "value not specified" end
		h.Health = math.clamp(v, 0, h.MaxHealth)
		return true, "Health = " .. tostring(h.Health)
	end,
	respawn = function()
		if not LocalPlayer then return false, "no player" end
		LocalPlayer:LoadCharacter()
		return true, "Character respawned"
	end,
}

local function htmlDecode(s)
	s = s:gsub("&amp;", "&"):gsub("&lt;", "<"):gsub("&gt;", ">")
	s = s:gsub("&quot;", "\""):gsub("&#x27;", "'"):gsub("&#39;", "'")
	s = s:gsub("&nbsp;", " "):gsub("&#x2F;", "/"):gsub("&hellip;", "...")
	return s
end
local function stripTags(s) return htmlDecode(s:gsub("<[^>]->", "")) end
local function decodeDDGUrl(href)
	if not href or href == "" then return "" end
	if href:sub(1,2) == "//" then href = "https:" .. href end
	local uddg = href:match("[?&]uddg=([^&]+)")
	if uddg then
		local ok, decoded = pcall(function() return HttpService:UrlDecode(uddg) end)
		if ok and decoded then return decoded end
	end
	return href
end

local function doHttpGet(url, headers)
	headers = headers or {}
	headers["User-Agent"] = headers["User-Agent"] or "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
	local data = { Url = url, Method = "GET", Headers = headers }
	if isStudio then
		return game.ReplicatedStorage.HTTP:InvokeServer(data)
	else
		if not http_func then error("No HTTP function available") end
		return http_func(data)
	end
end

local function webSearch(query)
	local results = {}
	if not query or query == "" then return results end
	local url = "https://html.duckduckgo.com/html/?q=" .. HttpService:UrlEncode(query)
	local ok, res = pcall(doHttpGet, url, {["Accept"] = "text/html"})
	if ok and res and res.Body then
		local body = res.Body
		for a in body:gmatch('<a[^>]-class="result__a"[^>]->.-</a>') do
			local href = a:match('href="([^"]+)"') or ""
			local title = a:match('>([^<]+)</a>') or a:match('>(.-)</a>') or ""
			local link = decodeDDGUrl(href)
			if link ~= "" and title ~= "" then
				table.insert(results, { title = stripTags(title), url = link, snippet = "" })
			end
		end
		local snippets = {}
		for sn in body:gmatch('<a[^>]-class="result__snippet"[^>]->(.-)</a>') do
			table.insert(snippets, stripTags(sn))
		end
		for i, s in ipairs(snippets) do
			if results[i] then results[i].snippet = s end
		end
	end
	if #results == 0 then
		local apiUrl = "https://api.duckduckgo.com/?q=" .. HttpService:UrlEncode(query) .. "&format=json&no_html=1&skip_disambig=1"
		local ok2, res2 = pcall(doHttpGet, apiUrl)
		if ok2 and res2 and res2.Body then
			local ok3, data = pcall(function() return HttpService:JSONDecode(res2.Body) end)
			if ok3 and data then
				if data.AbstractText and data.AbstractText ~= "" then
					table.insert(results, { title = data.Heading or query, url = data.AbstractURL or "", snippet = data.AbstractText })
				end
				if data.RelatedTopics then
					for _, t in ipairs(data.RelatedTopics) do
						if t.Text and t.FirstURL then
							table.insert(results, { title = t.Text:sub(1, 80), url = t.FirstURL, snippet = t.Text })
							if #results >= 5 then break end
						end
					end
				end
			end
		end
	end
	return results
end

local function formatSearchResults(results, limit)
	limit = limit or 6
	local lines = {}
	for i = 1, math.min(#results, limit) do
		local r = results[i]
		table.insert(lines, string.format("[%d] %s\n%s\n%s", i, r.title, r.snippet or "", r.url or ""))
	end
	return table.concat(lines, "\n\n")
end

local function richText(txt)
	local out = {}
	for line in (txt .. "\n"):gmatch("([^\n]*)\n") do
		local h, rest = line:match("^(#+)%s+(.*)$")
		if h and #h <= 6 then
			line = `<font size="{25 - #h * 2}"><b>{rest}</b></font>`
		end
		table.insert(out, line)
	end
	txt = table.concat(out, "\n")

	txt = txt:gsub("!%[([^%]]*)%]%(([^%)]+)%)", "<font color='#7FB8FF'>%1</font>")
	txt = txt:gsub("%[([^%]]+)%]%(([^%)]+)%)", "<font color='#7FB8FF'>%1</font>")
	txt = txt:gsub("<(https?://[^>]+)>", "<font color='#7FB8FF'>%1</font>")
	txt = txt:gsub("%*%*([^\n%*]+)%*%*", "<b>%1</b>")
	txt = txt:gsub("~~([^\n~]+)~~", "<strike>%1</strike>")
	txt = txt:gsub("`([^`\n]+)`", "<font color='#E8B86D'>%1</font>")
	txt = txt:gsub("%*([^\n%*]+)%*", "<i>%1</i>")

	return txt
end

-- ==================== MARKDOWN TABLE HELPERS ====================

local function stripEdgePipes(s)
	s = s:gsub("^%s*|", "")
	s = s:gsub("|%s*$", "")
	return s
end

local function splitCells(s)
	local cells = {}
	local current = ""
	for i = 1, #s do
		local ch = s:sub(i, i)
		if ch == "|" then
			table.insert(cells, current)
			current = ""
		else
			current = current .. ch
		end
	end
	table.insert(cells, current)
	for i = 1, #cells do
		cells[i] = cells[i]:gsub("^%s+", ""):gsub("%s+$", "")
	end
	return cells
end

-- ==================== MARKDOWN BLOCK PARSER ====================

local function isTableSeparator(line)
	if not line:match("^%s*|") then return false end
	local s = stripEdgePipes(line)
	local cells = splitCells(s)
	if #cells == 0 then return false end
	local valid = 0
	for _, t in ipairs(cells) do
		if t ~= "" then
			if not t:match("^:?-+:?$") then return false end
			valid = valid + 1
		end
	end
	return valid > 0
end

local function parseBlocks(text)
	local blocks = {}
	text = text:gsub("\r\n", "\n")
	local lines = {}
	for line in (text .. "\n"):gmatch("([^\n]*)\n") do
		table.insert(lines, line)
	end
	while #lines > 0 and lines[#lines] == "" do table.remove(lines) end

	local n = #lines
	local i = 1

	local function isCodeFence(l) return l:match("^```") ~= nil end
	local function isHR(l) return l:match("^%s*[-%*_][%-%*_][%-%*_]+%s*$") ~= nil end
	local function isTableRow(l) return l:match("^%s*|") ~= nil end
	local function isQuote(l) return l:match("^%s*>") ~= nil end
	local function isBullet(l) return l:match("^%s*[%-%*%+]%s") ~= nil end
	local function isNumber(l) return l:match("^%s*%d+%.%s") ~= nil end
	local function isList(l) return isBullet(l) or isNumber(l) end

	local function tryParseTable(startIdx)
		if startIdx + 1 > n then return nil end
		if not isTableRow(lines[startIdx]) then return nil end
		if not isTableSeparator(lines[startIdx + 1]) then return nil end
		local tblLines = {}
		local j = startIdx
		while j <= n and isTableRow(lines[j]) do
			table.insert(tblLines, lines[j])
			j = j + 1
		end
		if #tblLines < 2 then return nil end
		local header = splitCells(stripEdgePipes(tblLines[1]))
		if #header == 0 then return nil end
		local dataRows = {}
		for k = 3, #tblLines do
			table.insert(dataRows, splitCells(stripEdgePipes(tblLines[k])))
		end
		return { type = "table", header = header, rows = dataRows }, j
	end

	while i <= n do
		local line = lines[i]
		local consumed = false

		if isCodeFence(line) then
			local lang = line:match("^```(%S*)") or ""
			local code = {}
			i = i + 1
			while i <= n and not isCodeFence(lines[i]) do
				table.insert(code, lines[i])
				i = i + 1
			end
			i = i + 1
			table.insert(blocks, { type = "code", lang = lang, content = table.concat(code, "\n") })
			consumed = true
		elseif isHR(line) then
			table.insert(blocks, { type = "hr" })
			i = i + 1
			consumed = true
		else
			local tbl, nextI = tryParseTable(i)
			if tbl then
				table.insert(blocks, tbl)
				i = nextI
				consumed = true
			end
		end

		if not consumed then
			if isQuote(line) then
				local qLines = {}
				while i <= n and isQuote(lines[i]) do
					table.insert(qLines, lines[i]:gsub("^%s*>%s?", ""))
					i = i + 1
				end
				table.insert(blocks, { type = "quote", content = table.concat(qLines, "\n") })
			elseif isList(line) then
				local items = {}
				while i <= n and isList(lines[i]) do
					local l = lines[i]
					local indent = #(l:match("^%s*") or "")
					local bullet = l:match("^%s*[%-%*%+]%s") ~= nil
					local num = l:match("^%s*(%d+)%.%s")
					local content
					if bullet then
						content = l:match("^%s*[%-%*%+]%s+(.*)$") or ""
					else
						content = l:match("^%s*%d+%.%s+(.*)$") or ""
					end
					local taskState = nil
					local tm = content:match("^%[([ xX])%]%s*(.*)$")
					if tm then
						taskState = (tm:lower() == "x")
						content = content:match("^%[[ xX]%]%s*(.*)$") or ""
					end
					table.insert(items, {
						indent = indent,
						marker = bullet and "bullet" or "number",
						num = num,
						content = content,
						task = taskState
					})
					i = i + 1
				end
				table.insert(blocks, { type = "list", items = items })
			elseif line:match("^%s*$") then
				i = i + 1
			else
				local textLines = {}
				while i <= n do
					local l = lines[i]
					if l:match("^%s*$") or isCodeFence(l) or isHR(l) or isQuote(l) or isList(l) then break end
					local tb = tryParseTable(i)
					if tb then break end
					table.insert(textLines, l)
					i = i + 1
				end
				if #textLines > 0 then
					table.insert(blocks, { type = "text", content = table.concat(textLines, "\n") })
				else
					i = i + 1
				end
			end
		end
	end

	return blocks
end

local function measureTextWidth(text, size, font)
	if text == "" then return 0 end
	local ok, sz = pcall(function()
		return TextService:GetTextSize(text, size, font, Vector2.new(10000, 10000))

local function buildTable(parent, header, rows, layoutOrder)
	local colCount = #header
	for _, r in ipairs(rows) do
		if #r > colCount then colCount = #r end
	end
	if colCount == 0 then return end

	local FONT = Enum.Font.GothamMedium
	local FONT_BOLD = Enum.Font.GothamBold
	local TEXT_SIZE = 14
	local CELL_PADDING = 20

	local widths = {}
	for i = 1, colCount do widths[i] = 40 end
	local function update(col, txt)
		if not txt or txt == "" then return end
		local w = measureTextWidth(txt, TEXT_SIZE, FONT) + CELL_PADDING
		if w > widths[col] then widths[col] = w end
	end
	for i, cell in ipairs(header) do update(i, cell) end
	for _, row in ipairs(rows) do
		for i, cell in ipairs(row) do update(i, cell) end
	end

	local totalW = 0
	for i = 1, colCount do totalW = totalW + widths[i] end
	if totalW <= 0 then totalW = 1 end

	local frame = Instance.new("Frame")
	frame.Name = "TableBlock"
	frame.BackgroundColor3 = Color3.fromRGB(20, 20, 22)
	frame.BorderSizePixel = 0
	frame.Size = UDim2.new(1, 0, 0, 0)
	frame.AutomaticSize = Enum.AutomaticSize.Y
	frame.LayoutOrder = layoutOrder
	frame.Parent = parent

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, 6)
	corner.Parent = frame

	local stroke = Instance.new("UIStroke")
	stroke.Color = Color3.fromRGB(50, 50, 55)
	stroke.Thickness = 1
	stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	stroke.Parent = frame

	local pad = Instance.new("UIPadding")
	pad.PaddingTop = UDim.new(0, 6)
	pad.PaddingBottom = UDim.new(0, 6)
	pad.PaddingLeft = UDim.new(0, 6)
	pad.PaddingRight = UDim.new(0, 6)
	pad.Parent = frame

	local vLayout = Instance.new("UIListLayout")
	vLayout.FillDirection = Enum.FillDirection.Vertical
	vLayout.SortOrder = Enum.SortOrder.LayoutOrder
	vLayout.Padding = UDim.new(0, 0)
	vLayout.Parent = frame

	local rowOrder = 0
	local function makeRow(cells, isHeader)
		rowOrder = rowOrder + 1
		local rowFrame = Instance.new("Frame")
		rowFrame.BackgroundColor3 = isHeader and Color3.fromRGB(30, 30, 34) or Color3.fromRGB(0, 0, 0)
		rowFrame.BackgroundTransparency = isHeader and 0 or 1
		rowFrame.BorderSizePixel = 0
		rowFrame.Size = UDim2.new(1, 0, 0, 26)
		rowFrame.LayoutOrder = rowOrder
		rowFrame.Parent = frame

		local xScale = 0
		for i = 1, colCount do
			local text = cells[i] or ""
			local wScale = widths[i] / totalW

		lbl.RichText = true
		lbl.Text = prefix .. richText(item.content)
		lbl.Parent = rowFrame
	end

-- ==================== MD UTILS ====================

local function tryParseJSON(str)
	if not str or type(str) ~= "string" then return nil end
	local clean = str:gsub("```json", ""):gsub("```", "")
	local s = clean:find("{", 1, true)
	local e = clean:match(".*()}")
	if not s or not e then return nil end
	local ok, data = pcall(function() return HttpService:JSONDecode(clean:sub(s, e)) end)
	if ok and type(data) == "table" then return data end
	return nil
end

local function copy(text)
	if isStudio then print("[AI] Cannot copy in Studio.")
	elseif setclipboard then setclipboard(text)
	elseif toclipboard then toclipboard(text) end
end

-- ==================== LAYOUT ====================

local function relayout()
	local sorted = {}
	for _, c in ipairs(messagesList:GetChildren()) do
		if c:IsA("Frame") and c:GetAttribute("isMessage") then table.insert(sorted, c) end
	end
	table.sort(sorted, function(a, b) return (a:GetAttribute("msgOrder") or 0) < (b:GetAttribute("msgOrder") or 0) end)
	local y = 10
	for _, c in ipairs(sorted) do
		c.Position = UDim2.new(c.AnchorPoint.X, 0, 0, y)
		y += c.AbsoluteSize.Y + 10
	end
	messagesList.CanvasSize = UDim2.new(0, 0, 0, y + 10)
end

local function scheduleRelayout()
	task.defer(relayout)
	task.delay(0.1, relayout)
	task.delay(0.4, relayout)
end

local function renderAssistantContent(Clone, Message)
	local container = Clone.Message
	local textTemplate = container.SysMessageText
	local codeTemplate = container.CodeBlock
	for _, c in ipairs(container:GetChildren()) do
		if c ~= textTemplate and c ~= codeTemplate and c:IsA("GuiObject") then c:Destroy() end
	end
	textTemplate.Visible = false
	codeTemplate.Visible = false

	local blocks = parseBlocks(Message)
	local order = 0

	local function addTextBlock(txt)
		local t = txt:gsub("^%s+", ""):gsub("%s+$", "")
		if #t == 0 then return end
		local lbl = textTemplate:Clone()
		lbl.Name = "TextBlock"
		lbl.Visible = true
		lbl.LayoutOrder = order
		lbl.Text = richText(t)
		lbl.Parent = container
		order = order + 1
	end

	for _, block in ipairs(blocks) do
		if block.type == "text" then
			addTextBlock(block.content)
		elseif block.type == "code" then
			local cb = codeTemplate:Clone()
			cb.Name = "CodeBlockInstance"
			cb.Visible = true
			cb.LayoutOrder = order
			cb.Parent = container
			cb.CodeHeader.CodeLang.Text = (#block.lang > 0) and block.lang:lower() or "code"
			cb.CodeBox.CodeText.Text = block.content
			local btn = cb.CodeHeader.CodeCopyBtn
			btn.MouseButton1Click:Connect(function()
				copy(block.content)
				btn.Text = "Copied!"
				task.delay(1.2, function() if btn.Parent then btn.Text = "Copy" end end)
			end)
			order = order + 1
		elseif block.type == "hr" then
			local hr = Instance.new("Frame")
			hr.Name = "HRBlock"
			hr.BackgroundColor3 = Color3.fromRGB(50, 50, 55)
			hr.BorderSizePixel = 0
			hr.Size = UDim2.new(1, 0, 0, 1)
			hr.LayoutOrder = order
			hr.Parent = container
			order = order + 1
		elseif block.type == "quote" then
			local ok, err = pcall(buildQuote, container, block.content, order)
			if not ok then addTextBlock("> " .. block.content) end
			order = order + 1
		elseif block.type == "table" then
			local ok, err = pcall(buildTable, container, block.header, block.rows, order)
			if not ok then
				local flat = {}
				table.insert(flat, table.concat(block.header, " | "))
				for _, r in ipairs(block.rows) do
					table.insert(flat, table.concat(r, " | "))
				end
				addTextBlock(table.concat(flat, "\n"))
			end
			order = order + 1
		elseif block.type == "list" then
			local ok, err = pcall(buildList, container, block.items, order)
			if not ok then
				local flat = {}
				for _, item in ipairs(block.items) do
					local indentStr = string.rep("  ", math.floor(item.indent / 2))
					local mark
					if item.task ~= nil then
						mark = item.task and "[x] " or "[ ] "
					elseif item.marker == "number" then
						mark = (item.num or "1") .. ". "
					else
						mark = "- "
					end
					table.insert(flat, indentStr .. mark .. item.content)
				end
				addTextBlock(table.concat(flat, "\n"))
			end
			order = order + 1
		end
	end
end

-- ==================== API ====================

local function callGroq(payload, temperature, maxTokens)
	local Data = {
		Url = GROQ_URL,
		Method = "POST",
		Headers = {
			["Content-Type"] = "application/json",
			["Authorization"] = "Bearer " .. GROQ_API_KEY
		},
		Body = {
			model = GROQ_MODEL,
			messages = payload,
			temperature = temperature or 0.7,
			max_tokens = maxTokens or 1500,
			top_p = 1,
			stream = false
		}
	}
	local Result
	local ok, err = pcall(function()
		if isStudio then
			Result = game.ReplicatedStorage.HTTP:InvokeServer(Data)
		else
			if not http_func then error("No HTTP function available") end
			Data.Body = HttpService:JSONEncode(Data.Body)
			local raw = http_func(Data)
			Result = HttpService:JSONDecode(raw.Body)
		end
	end)
	if not ok then return nil, tostring(err) end
	if Result and Result.error then return nil, tostring(Result.error.message or Result.error) end
	if Result and Result.choices and Result.choices[1] and Result.choices[1].message then
		return Result.choices[1].message.content, nil
	end
	return nil, "Empty API response"
end

-- ==================== MEMORY OPS ====================

local function buildMemorySystemMessage()
	if memory.summary == "" and #memory.facts == 0 then return nil end
	local factsStr = #memory.facts > 0 and ("- " .. table.concat(memory.facts, "\n- ")) or "(none yet)"
	local summaryStr = memory.summary ~= "" and memory.summary or "(no summary yet)"
	local text = string.format(MEMORY_SYSTEM_TEMPLATE, summaryStr, factsStr)
	return { role = "system", content = text }
end

local function compressHistory()
	if #memory.recent <= COMPRESS_AT then return end
	local toCompress = {}
	for i = 1, #memory.recent - COMPRESS_KEEP do table.insert(toCompress, memory.recent[i]) end
	local keep = {}
	for i = #memory.recent - COMPRESS_KEEP + 1, #memory.recent do table.insert(keep, memory.recent[i]) end
	local dialogText = {}
	for _, m in ipairs(toCompress) do
		local prefix = m.role == "user" and "USER: " or (m.role == "assistant" and "AI: " or (tostring(m.role):upper() .. ": "))
		table.insert(dialogText, prefix .. tostring(m.content))
	end
	local payload = {
		{ role = "system", content = COMPRESS_PROMPT },
		{ role = "user", content = "PREVIOUS SUMMARY:\n" .. (memory.summary ~= "" and memory.summary or "(empty)") ..
			"\n\nNEW DIALOGUE:\n" .. table.concat(dialogText, "\n") }
	}
	local answer = callGroq(payload, 0.2, 500)
	if answer and answer ~= "" then
		memory.summary = answer:gsub("^%s+", ""):gsub("%s+$", "")
	end
	memory.recent = keep
	saveMemory()
end

local function extractFacts()
	local chunk = {}
	local from = math.max(1, #memory.recent - 6)
	for i = from, #memory.recent do
		table.insert(chunk, tostring(memory.recent[i].role):upper() .. ": " .. tostring(memory.recent[i].content))
	end
	if #chunk == 0 then return end
	local payload = {
		{ role = "system", content = MEMORY_EXTRACT_PROMPT },
		{ role = "user", content = table.concat(chunk, "\n") }
	}
	local answer = callGroq(payload, 0.1, 250)
	if not answer then return end
	local parsed = tryParseJSON(answer)
	if parsed and type(parsed.facts) == "table" then
		local existing = {}
		for _, f in ipairs(memory.facts) do existing[f:lower()] = true end
		for _, f in ipairs(parsed.facts) do
			if type(f) == "string" and #f > 2 and not existing[f:lower()] then
				table.insert(memory.facts, f)
				if #memory.facts > 40 then table.remove(memory.facts, 1) end
			end
		end
		saveMemory()
	end
end

local function appendExchange(userMsg, assistantMsg)
	table.insert(memory.recent, { role = "user", content = userMsg })
	table.insert(memory.recent, { role = "assistant", content = assistantMsg })
	saveMemory()
	task.spawn(function()
		pcall(compressHistory)
		pcall(extractFacts)
	end)
end

local function buildPayload(extraSystem, useHistory)
	local payload = { { role = "system", content = BASE_SYSTEM } }
	local memSys = buildMemorySystemMessage()

	if extraSystem then table.insert(payload, { role = "system", content = extraSystem }) end

		for _, m in ipairs(memory.recent) do table.insert(payload, m) end
	end
	return payload
end

-- ==================== THINK ====================

local function runThink(query, Response)
	setResponseStatus(Response, "Analyzing the request...")
	local routerPayload = buildPayload(THINK_ROUTER_PROMPT, false)
	table.insert(routerPayload, { role = "user", content = query })

	local routeStr, routeErr = callGroq(routerPayload, 0.1, 300)
	if not routeStr then return "Router error: " .. tostring(routeErr) end
	local route = tryParseJSON(routeStr) or { needs_search = false, search_query = "", plan = "" }

	local searchContext = ""
	local sourcesList = {}

	if route.needs_search and route.search_query and route.search_query ~= "" then
		setResponseStatus(Response, "Searching: " .. route.search_query)
		local results = webSearch(route.search_query)
		if #results > 0 then
			local more = webSearch(query .. " details")
			for _, r in ipairs(more) do
				local dup = false
				for _, existing in ipairs(results) do
					if existing.url == r.url then dup = true break end
				end
				if not dup then table.insert(results, r) end
				if #results >= 10 then break end
			end
		end
		if #results > 0 then
			searchContext = formatSearchResults(results, 8)
			for i = 1, math.min(#results, 4) do
				table.insert(sourcesList, string.format("[%d] %s - %s", i, results[i].title, results[i].url))
			end
			setResponseStatus(Response, string.format("Found %d sources, thinking deeply...", #results))
		else
			setResponseStatus(Response, "No web results, answering from knowledge...")
		end
	else
		setResponseStatus(Response, "Thinking deeply...")
	end

	local answerPayload = buildPayload(THINK_ANSWER_PROMPT, false)
	if searchContext ~= "" then
		table.insert(answerPayload, {
			role = "system",
			content = 'Web search results for "' .. tostring(route.search_query) .. '":\n\n' .. searchContext
		})
	end
	if route.plan and route.plan ~= "" then
		table.insert(answerPayload, { role = "system", content = "Your plan: " .. tostring(route.plan) })
	end
	for _, m in ipairs(memory.recent) do table.insert(answerPayload, m) end
	table.insert(answerPayload, { role = "user", content = query })

	local answer, err = callGroq(answerPayload, 0.35, 4000)
	if not answer then return "Think error: " .. tostring(err) end
	local final = answer
	if #sourcesList > 0 then
		final = final .. "\n\n**Sources:**\n" .. table.concat(sourcesList, "\n")
	end
	return final
end

-- ==================== MEMORY COMMANDS ====================

local function memoryReport()
	local lines = {}
	table.insert(lines, "**Memory**")
	table.insert(lines, "Messages in session: " .. #memory.recent)
	table.insert(lines, "File: `" .. MEMORY_FILE .. "` " .. (hasFileIO and "(persistence available)" or "(no writefile - session only)"))
	if memory.summary ~= "" then table.insert(lines, "\n**Summary:**\n" .. memory.summary) end
	if #memory.facts > 0 then table.insert(lines, "\n**Facts:**\n- " .. table.concat(memory.facts, "\n- ")) end
	if #memory.recent > 0 then
		table.insert(lines, "\n**Recent messages:**")
		for i, m in ipairs(memory.recent) do
			local role = m.role == "user" and "**User**" or (m.role == "assistant" and "**AI**" or "**sys**")
			local text = tostring(m.content):gsub("\n", " ")
			if #text > 120 then text = text:sub(1, 117) .. "..." end
			table.insert(lines, string.format("%d. %s: %s", i, role, text))
		end
	end
	return table.concat(lines, "\n")
end

local function forgetAll()
	memory.summary = ""
	memory.facts = {}
	memory.recent = {}
	if hasFileIO then pcall(function() if isfile(MEMORY_FILE) then delfile(MEMORY_FILE) end end) end
	return "Memory fully cleared."
end

local function forgetLast()
	if #memory.recent >= 2 then
		table.remove(memory.recent)
		table.remove(memory.recent)
		saveMemory()
		return "Removed last exchange."
	end
	return "Nothing to remove."
end

-- ==================== SEND ====================

local function sendMessage()
	if isGenerating then return end
	local Prompt = box.Text
	if Prompt == nil or Prompt == "" then return end
	box.Text = ""

	local lower = Prompt:lower()
	local isChCmd = (lower:sub(1, 4) == "@ch ") or (lower == "@ch")
	local isServerCmd = (lower:sub(1, 8) == "@server ") or (lower == "@server")
	local isSearchCmd = (lower:sub(1, 8) == "@search ") or (lower == "@search")
	local isThinkCmd = (lower:sub(1, 7) == "@think ") or (lower == "@think")
	local isMemoryCmd = (lower == "@memory")
	local isForgetCmd = (lower:sub(1, 7) == "@forget") or (lower == "@forget")
	local isClearCmd = (lower == "@clear")
	local isToolCmd = isChCmd or isServerCmd

	if isClearCmd then clearChat() return end

	if isMemoryCmd then
		createMessage(true, Prompt)
		local r = createMessage(false, memoryReport())
		r:SetAttribute("rawText", memoryReport())
		return
	end
	if isForgetCmd then
		createMessage(true, Prompt)
		local sub = lower:match("^@forget%s+(%S+)")
		local r
		if sub == "last" then r = forgetLast() else r = forgetAll() end
		createMessage(false, r)
		return
	end

	createMessage(true, Prompt)
	isGenerating = true
	local Response = createMessage(false, "Thinking...")

	task.spawn(function()
		local Dots = 3
		while task.wait(.33) do
			if not isGenerating then return end
			Dots = (Dots % 3) + 1
			setResponseStatus(Response, "Thinking" .. string.rep(".", Dots))
		end
	end)

	task.spawn(function()
		local finalDisplay, historyContent

		if isThinkCmd then
			local query = Prompt:sub(8):gsub("^%s+", "")
			if query == "" then
				isGenerating = false
				setResponseStatus(Response, "Provide a query: `@think <what to think about>`")
				return
			end
			local answer = runThink(query, Response)
			isGenerating = false
			finalDisplay = answer
			historyContent = answer

		elseif isSearchCmd then
			local query = Prompt:sub(9):gsub("^%s+", "")
			if query == "" then
				isGenerating = false
				setResponseStatus(Response, "Provide a query: `@search <what to search>`")
				return
			end
			setResponseStatus(Response, "Searching the web: " .. query)
			local results = webSearch(query)
			if #results == 0 then
				isGenerating = false
				finalDisplay = "Nothing found for: `" .. query .. "`."
				historyContent = finalDisplay
			else
				setResponseStatus(Response, string.format("Found %d results, processing...", #results))
				local resultsText = formatSearchResults(results, 6)
				local payload = buildPayload(SEARCH_PROMPT, true)
				table.insert(payload, { role = "system", content = "Web search results for: \"" .. query .. "\"\n\n" .. resultsText })
				table.insert(payload, { role = "user", content = query })
				local answer, err = callGroq(payload, 0.4, 1500)
				isGenerating = false
				if not answer then
					finalDisplay = "Search error: " .. tostring(err)
				else
					local sources = {}
					for i = 1, math.min(#results, 3) do
						table.insert(sources, string.format("[%d] %s - %s", i, results[i].title, results[i].url))
					end
					finalDisplay = answer .. "\n\n**Sources:**\n" .. table.concat(sources, "\n")
				end
				historyContent = finalDisplay
			end

		else
			local extraSys
			if isServerCmd then extraSys = TOOL_PROMPT_SERVER
			elseif isChCmd then extraSys = TOOL_PROMPT_CH end

			local payload = buildPayload(extraSys, true)
			table.insert(payload, { role = "user", content = Prompt })

			local answer, err = callGroq(payload, isToolCmd and 0.2 or 0.7, 1500)
			isGenerating = false

			local Msg = answer or ("Request error: " .. tostring(err))
			local displayMsg = Msg
			local historyMsg = Msg

			if isToolCmd then
				local parsed = tryParseJSON(Msg)
				if parsed and type(parsed) == "table" then
					local reply = parsed.reply
					if type(reply) ~= "string" or reply == "" then reply = "Done." end
					local results = {}
					if type(parsed.actions) == "table" then
						for _, action in ipairs(parsed.actions) do
							local aok, amsg = executeAction(action, isServerCmd)
							table.insert(results, (aok and "OK " or "FAIL ") .. amsg)
						end
					end
					local final = reply
					if #results > 0 then final = reply .. "\n\n" .. table.concat(results, "\n") end
					displayMsg = final
					historyMsg = reply
				end
			end
			finalDisplay = displayMsg
			historyContent = historyMsg
		end

		isGenerating = false
		appendExchange(Prompt, historyContent)
		Response:SetAttribute("rawText", finalDisplay)
		renderAssistantContent(Response, finalDisplay)
		scheduleRelayout()
	end)
end

	if enterPressed then sendMessage() end
end)

task.spawn(function()
	loadMemory()
	if memory.summary ~= "" or #memory.facts > 0 or #memory.recent > 0 then
		local welcome = "**Memory loaded.**"
		if memory.summary ~= "" then welcome = welcome .. "\n- Saved summary available." end
		if #memory.facts > 0 then welcome = welcome .. "\n- Facts about you: " .. #memory.facts end
		if #memory.recent > 0 then welcome = welcome .. "\n- Session messages: " .. #memory.recent end
		welcome = welcome .. "\n\nType `@memory` for details, `@forget` to clear, `@clear` to clear the chat window."
		createMessage(false, welcome)
	end
end)
